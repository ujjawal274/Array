Best time to buy and sell stock

SOURCE CODE:

class Solution:
    def maxProfit(Self,prices,target):
        min_price = prices[0]
        max_profit = 0

        for i in range (1,len(prices)):
            current_profit= prices[1] - min_price 

            if current_profit > max_profit:
                max_profit = current_profit

            if prices[i] < min_price:
                min_price = prices[i]

        return max_profit
